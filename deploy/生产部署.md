# ============================================================
#  生产部署配置 —— 聊天室
#  适用于：公网云服务器（Ubuntu/Debian）+ Nginx + Let's Encrypt
# ============================================================

# ---------- 1. 上传代码 ----------

在服务器上：
```bash
mkdir -p /opt/chat-room
cd /opt/chat-room
# 把项目文件上传到这里（scp / 拖拽 / git clone）

# ---------- 2. 安装依赖并试跑 ----------

# 需要 Node.js >= 22（用到内置的 node:sqlite）
node --version

npm install --omit=dev

# 先手动跑一次确认能起来
ALT_PORT=0 PORT=8000 node server.js
# 看到「聊天室已启动 → http://localhost:8000」即正常，Ctrl+C 停掉

# ---------- 3. 用 systemd 常驻 ----------

cat > /etc/systemd/system/chat-room.service <<'EOF'
[Unit]
Description=Chat Room WebSocket Server
After=network.target

[Service]
Type=simple
User=www-data
WorkingDirectory=/opt/chat-room
Environment=PORT=8000
Environment=HOST=0.0.0.0
Environment=ALT_PORT=0
Environment=NODE_ENV=production
ExecStart=/usr/bin/node server.js
Restart=always
RestartSec=5

# 加固
NoNewPrivileges=true
PrivateTmp=true
ProtectSystem=full
ReadWritePaths=/opt/chat-room/data

[Install]
WantedBy=multi-user.target
EOF

chmod 600 /opt/chat-room/data/chat.db
chown -R www-data:www-data /opt/chat-room

systemctl daemon-reload
systemctl enable --now chat-room
systemctl status chat-room

# ---------- 4. Nginx 反代（含 WebSocket 升级） ----------

cat > /etc/nginx/sites-available/chat-room <<'EOF'
server {
    listen 80;
    server_name chat.example.com;      # 改成你的域名

    # WebSocket 升级必需
    location /ws {
        proxy_pass http://127.0.0.1:8000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # 长连接不能被过早掐断
        proxy_read_timeout 3600s;
        proxy_send_timeout 3600s;
    }

    location / {
        proxy_pass http://127.0.0.1:8000;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}
EOF

ln -sf /etc/nginx/sites-available/chat-room /etc/nginx/sites-enabled/
nginx -t && systemctl reload nginx

# ---------- 5. HTTPS（必须是 wss:// 的前提） ----------

apt install certbot python3-certbot-nginx
certbot --nginx -d chat.example.com

# certbot 会自动改写 nginx 配置，加上：
#   listen 443 ssl;
#   location /ws { ... proxy_set_header Connection "upgrade"; ... }
# 验证 wss 生效：
#   curl -i -N -H "Connection: Upgrade" -H "Upgrade: websocket" \
#     -H "Sec-WebSocket-Version: 13" -H "Sec-WebSocket-Key: x3JJHMbDL1EzLkh9GBhXDw==" \
#     https://chat.example.com/ws

# ---------- 6. 防火墙 ----------
ufw allow 80/tcp
ufw allow 443/tcp
# 8000 只监听 127.0.0.1，不要对公网开放

# ---------- 7. 数据备份 ----------
# SQLite 单文件，备份就是拷一个文件
cat > /opt/backup-chat.sh <<'EOF'
#!/bin/bash
mkdir -p /opt/chat-room/backups
cd /opt/chat-room/data
sqlite3 chat.db ".backup 'backups/chat-$(date +%F).db'" 2>/dev/null \
  || cp chat.db "backups/chat-$(date +%F).db"
find backups -name "chat-*.db" -mtime +14 -delete
EOF
chmod +x /opt/backup-chat.sh
# 加到 crontab：每天凌晨 3 点
# 0 3 * * * /opt/backup-chat.sh

# ============================================================
#  常见问题
# ============================================================

# Q: 页面能打开但消息发不出去？
#   A: Nginx 少了 WebSocket 升级配置。检查 location /ws 段。
#      浏览器 F12 → Network → WS → 状态必须是 101 Switching Protocols。

# Q: 浏览器提示"混合内容"？
#   A: 页面是 https 但 WS 连的是 ws://。检查 app.js 的 proto 判断，
#      正常会自动切 wss://，前提是 Nginx 正确透传了 X-Forwarded-Proto。

# Q: 重启后消息没了？
#   A: DATA_DIR 指向了临时目录。挂载持久化卷，或确认 data/ 在持久化路径上。

# Q: 朋友还是打不开？
#   A: 1) 域名 DNS 是否已解析到服务器 IP
#      2) 云厂商安全组是否放行 80/443
#      3) ufw 是否放行
#      4) certbot 是否成功
